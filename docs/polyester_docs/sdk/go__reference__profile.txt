updates, err := client.Auth.Profile.SubscribeIdentity(ctx)
if err != nil { log.Fatal(err) }
defer updates.Close()

for identity := range updates.Messages() {
	fmt.Println(
		identity.AccountID,
		identity.Username,
		identity.AvatarURL,
		identity.RootSmartAccountAddress,
	)
	break
}
