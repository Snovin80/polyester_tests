from polyester import AsyncPolyester, POLYESTER_TESTNET_ENVIRONMENT

async with AsyncPolyester(
    environment=POLYESTER_TESTNET_ENVIRONMENT,
    api_key_id="ak_...",
    api_private_key="...",  # 64-char hex from key creation
    default_account_id="...",  # Profile → Account ID
    # api_url=None,  # overrides environment.api_url
    # ws_url=None,  # overrides environment.websocket_url
    # default_sub_account_id=None,
    # timeout=10.0,
    # wire_format="binary",  # or "json"
    # hydrate_catalogs=True,
) as client:
    await client.wait_for_catalogs()
