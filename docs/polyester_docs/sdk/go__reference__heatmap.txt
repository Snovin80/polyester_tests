symbol := "BTC-USDT"
history, err := client.Heatmap.Get(
	ctx, &symbol, nil,
	"1s", 50, 100, "close",
	nil, nil, nil,
)
if err != nil { log.Fatal(err) }
fmt.Println(history.Raw)

---

sub, err := client.Heatmap.SubscribeLive(ctx, &symbol, nil, "1s")
if err != nil { log.Fatal(err) }
defer sub.Close()
for bucket := range sub.Messages() {
	fmt.Println(bucket.Raw)
	break
}
