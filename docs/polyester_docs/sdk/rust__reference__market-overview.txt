=== ПРИМЕРЫ (дословно) ===

use polyester::services::ListMarketOverviewOptions;

let overview = client
    .market_overview
    .list(ListMarketOverviewOptions {
        symbols: Some(vec!["BTC-USDT".into()]),
        limit: Some(50),
        include_sparklines: false,
        ..Default::default()
    })
    .await?;
for m in &overview.markets {
    println!("{} {} {:?}", m.symbol_id, m.symbol, m.last_price);
}

---

use polyester::services::MarketOverviewCreateSubscriptionOptions;

let mut sub = client
    .market_overview
    .create_subscription(MarketOverviewCreateSubscriptionOptions {
        symbols: Some(vec!["BTC-USDT".into()]),
        limit: Some(50),
        ..Default::default()
    })
    .await?;

while let Some(rows) = sub.updates().recv().await {
    for m in &rows {
        println!("{} {:?}", m.symbol, m.last_price);
    }
    break;
}
sub.close();
