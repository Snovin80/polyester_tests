await client.wait_for_catalogs()

history = await client.heatmap.get(
    symbol="BTC-USDT",
    interval="1s",
    depth=50,
    limit=100,
    quantity_mode="close",
)
print(history.raw)

---

sub = await client.heatmap.subscribe_live(
    symbol="BTC-USDT",
    interval="1s",
)
async with sub:
    async for bucket in sub:
        print(bucket.raw)
        break
