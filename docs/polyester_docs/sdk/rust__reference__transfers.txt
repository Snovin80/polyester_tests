=== ПРИМЕРЫ (дословно) ===

use polyester::proto::ledger::read::v1::ListTransfersRequest;

let page = client.transfers.list(ListTransfersRequest {
    limit: 50,
    ..Default::default()
}).await?;
for transfer in page.transfers {
    println!(
        "{} {} {} debit={}",
        transfer.tx_id,
        transfer.asset_id,
        transfer.amount,
        transfer.is_debit,
    );
}

---

let mut sub = client.transfers.subscribe(None).await?;
if let Some(transfer) = sub.recv_result().await? {
    println!("{} {}", transfer.tx_id, transfer.amount);
}
