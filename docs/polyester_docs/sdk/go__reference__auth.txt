sub, err := client.Auth.Profile.SubscribeIdentity(ctx)
if err != nil { log.Fatal(err) }
defer sub.Close()
for identity := range sub.Messages() {
	fmt.Println(identity.AccountID, identity.Username, identity.AvatarURL)
	break
}
