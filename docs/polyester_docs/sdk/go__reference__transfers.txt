page, err := client.Transfers.List(ctx, nil, nil, 50, true, nil)
if err != nil { log.Fatal(err) }
for _, transfer := range page.Transfers {
	fmt.Println(
		transfer.TxID,
		transfer.AssetID,
		transfer.Amount,
		transfer.IsDebit,
		transfer.Pending,
	)
}

if page.NextCursor != nil {
	next, err := client.Transfers.List(ctx, nil, nil, 50, false, page.NextCursor)
	_ = next
	_ = err
}

---

sub, err := client.Transfers.Subscribe(ctx, nil) // nil → DefaultAccountID
if err != nil { log.Fatal(err) }
defer sub.Close()
for transfer := range sub.Messages() {
	fmt.Println(transfer.TxID, transfer.Amount)
	break
}
